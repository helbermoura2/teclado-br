cmake_minimum_required(VERSION 3.22)
project(cenallm LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
foreach(flag LLAMA_BUILD_TESTS LLAMA_BUILD_TOOLS LLAMA_BUILD_EXAMPLES LLAMA_BUILD_SERVER LLAMA_BUILD_APP LLAMA_OPENSSL LLAMA_SUBPROCESS GGML_OPENMP GGML_NATIVE GGML_BACKEND_DL GGML_LLAMAFILE)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
if(NOT LLAMA_SOURCE_DIR)
  message(FATAL_ERROR "Set LLAMA_SOURCE_DIR to the pinned llama.cpp checkout")
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" llama-build)
add_library(cenallm SHARED local_ai.cpp)
target_link_libraries(cenallm PRIVATE llama-common llama log)
target_link_options(cenallm PRIVATE "-Wl,-z,max-page-size=16384")
